# Rules for the OLD scripts, services pipeline on cp-mint-svcplanted (moved from the 2026-10-21 svcpipe.sh):
# baseline, systemconfig, then every service script with defaults + the README answers the checks use.
set HARD=900

step s1-baseline bash services/baseline.sh

step s3-sysconfig bash systemconfig.sh
rule Proceed with operation             => y
rule Open or Close a port               => n
rule Enable IPv6                        => n
rule install stable updates             => yes
rule Do you want to continue            => y
rule Remove ALL superuser               => n
rule Lock the root account              => y
rule Linux Mint: turn on Update Manager => y

step s-ssh bash services/ssh.sh
